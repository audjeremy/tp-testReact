export function average(values) {
  if (!Array.isArray(values) || values.length === 0) return 0
  let total = 0
  let count = 0
  for (const v of values) {
    if (Number.isFinite(v)) {
      total += v
      count += 1
    }
  }
  return count === 0 ? 0 : total / count
}
