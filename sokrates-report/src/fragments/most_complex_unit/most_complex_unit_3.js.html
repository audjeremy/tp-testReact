export function formatName(first, last) {
  const safe = (s) =>
    (typeof s === 'string' ? s : '')
      .trim()
      .replace(/\s+/g, ' ')
      .toLowerCase()
      .replace(/^\p{Letter}/u, (ch) => ch.toUpperCase()) // capitalise 1re lettre (unicode)
  const f = safe(first)
  const l = safe(last)
  if (f && l) return `${f} ${l}`
  if (f) return f
  if (l) return l
  return ''
}
